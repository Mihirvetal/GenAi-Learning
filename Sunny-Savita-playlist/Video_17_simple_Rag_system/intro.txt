import weaviate
import os
import re
from weaviate.classes.init import Auth
from google.colab import userdata

# ==========================================
# 2. DATABASE CONNECTION
# ==========================================
WEAVIATE_API_KEY = "ADD_FROM_DOTENV"
WEAVIATE_CLUSTER_URL = "ADD_FROM_DOTENV"
# Connect to Weaviate Cloud
client = weaviate.connect_to_weaviate_cloud(
    cluster_url=WEAVIATE_CLUSTER_URL,
    auth_credentials=Auth.api_key(WEAVIATE_API_KEY),
)
print(f"Weaviate Connection Status: {client.is_ready()}")

# ==========================================
# 3. EMBEDDINGS & DATA INGESTION
# ==========================================
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import CharacterTextSplitter

# Load Embeddings
embeddings = HuggingFaceEmbeddings(
    model_name='sentence-transformers/all-mpnet-base-v2',
    model_kwargs={'device': 'cpu'}
)

# Load PDF
pdf_url = "https://arxiv.org/pdf/2306.15217"
loader = PyPDFLoader(pdf_url, extract_images=True)
pages = loader.load()

# Split Text
text_splitter = CharacterTextSplitter(chunk_size=5000, chunk_overlap=100)
texts = text_splitter.split_documents(pages)

# Sanitize Metadata
for doc in texts:
    clean_metadata = {}
    for key, value in doc.metadata.items():
        clean_key = re.sub(r'[^_A-Za-z0-9]', '_', key)
        clean_metadata[clean_key] = value
    doc.metadata = clean_metadata

# ==========================================
# 4. VECTOR DATABASE STORAGE
# ==========================================
from langchain_weaviate import WeaviateVectorStore

# Clear old sandbox data to free up the 1-collection limit
client.collections.delete_all()
print("Cloud database cleared! Your 1-collection slot is now free. 🟢")

# Store chunks
vector_db = WeaviateVectorStore.from_documents(
    documents=texts,
    embedding=embeddings,
    client=client,
    index_name="ArxivPaperRAG"
)
print("All chunks successfully embedded and stored in Weaviate Cloud! 🚀")

# ==========================================
# 5. GROQ LLM INITIALIZATION
# ==========================================
from langchain_groq import ChatGroq
import os

# Get Groq API key from Colab Secrets
groq_api = userdata.get('groq_api')
os.environ["GROQ_API_KEY"] = groq_api

# Initialize the Groq LLM with a currently active model!
llm = ChatGroq(
    model="llama-3.1-8b-instant",  # 👈 CHANGED THIS LINE
    temperature=0.7
)
print("Groq LLM successfully loaded with Llama 3.1! 🧠")








# ==========================================
# 6. LCEL RAG CHAIN & EXECUTION
# ==========================================
from langchain_core.runnables import RunnablePassthrough
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate

# Create Retriever
retriever = vector_db.as_retriever(search_kwargs={"k": 3})

# Define Prompt
template = """You are an expert AI assistant. Answer the question based ONLY on the following context:
{context}

Question: {question}
Answer:"""

prompt = ChatPromptTemplate.from_template(template)

# Build Pipeline
chain = (
    {"context": retriever, "question": RunnablePassthrough()}
    | prompt
    | llm
    | StrOutputParser()
)
print("LCEL Chain successfully built and ready for queries! ⛓️")

# Run Inference
user_question = "What are the main findings of the paper regarding context engineering?"
response = chain.invoke(user_question)

print("\n🤖 AI Response:\n")
print(response)
